</head>
<body class="bg-slate-950 text-slate-100 min-h-screen font-sans flex flex-col items-center justify-center p-4">

    <div class="max-w-md w-full bg-slate-900 border border-slate-800 rounded-2xl p-6 shadow-2xl">
        <!-- Header -->
        <div class="text-center mb-6">
            <h1 class="text-2xl font-bold bg-gradient-to-r from-blue-400 to-indigo-500 bg-clip-text text-transparent">Truth Guardian 🛡️</h1>
            <p class="text-xs text-slate-400 mt-1">Say No to Clickbait & Fake Content</p>
        </div>

        <!-- Mode Selection Gateway -->
        <div id="gateway" class="space-y-4">
            <p class="text-sm text-center text-slate-300 font-medium">Aap aaj kis mode mein hain?</p>
            <button onclick="selectMode('creator')" class="w-full py-3 px-4 bg-blue-600 hover:bg-blue-500 rounded-xl font-semibold transition shadow-lg flex items-center justify-center gap-2 cursor-pointer">
                🚀 Main Creator Hoon (Ideas & SEO)
            </button>
            <button onclick="selectMode('viewer')" class="w-full py-3 px-4 bg-indigo-600 hover:bg-indigo-500 rounded-xl font-semibold transition shadow-lg flex items-center justify-center gap-2 cursor-pointer">
                🛡️ Main Viewer Hoon (Fake/Scam Detector)
            </button>
        </div>

        <!-- Dynamic Workspace (Hidden by default, zero bugs guaranteed) -->
        <div id="workspace" class="hidden space-y-4 mt-4">
            <div id="workspace-content"></div>
            <button onclick="resetApp()" class="w-full py-2 text-xs text-slate-400 hover:text-white transition mt-4 underline cursor-pointer">← Back to Gateway</button>
        </div>
    </div>

    <script>
        function selectMode(mode) {
            document.getElementById('gateway').classList.add('hidden');
            const workspace = document.getElementById('workspace');
            const content = document.getElementById('workspace-content');
            workspace.classList.remove('hidden');

            if (mode === 'creator') {
                content.innerHTML = `
                    <h2 class="text-lg font-bold text-blue-400">Creator Studio</h2>
                    <p class="text-xs text-slate-400">Apna topic ya niche dalein (e.g., Science, Tech, Earn Money):</p>
                    <input type="text" id="userInput" placeholder="Type topic here..." class="w-full bg-slate-950 border border-slate-800 rounded-lg p-3 text-sm text-white focus:outline-none focus:border-blue-500">
                    <button onclick="generateCreatorData()" class="w-full py-2.5 bg-blue-600 hover:bg-blue-500 rounded-lg font-medium text-sm transition cursor-pointer">Generate Hit Idea & SEO</button>
                    <div id="result" class="text-xs bg-slate-950 p-3 rounded-lg border border-slate-800 hidden mt-2 text-slate-300"></div>
                `;
            } else {
                content.innerHTML = `
                    <h2 class="text-lg font-bold text-indigo-400">Viewer Truth Detector</h2>
                    <p class="text-xs text-slate-400">Video ka title ya link paste karein (check karne ke liye ke fake hai ya real):</p>
                    <input type="text" id="userInput" placeholder="Paste video title or topic..." class="w-full bg-slate-950 border border-slate-800 rounded-lg p-3 text-sm text-white focus:outline-none focus:border-indigo-500">
                    <button onclick="detectFakeContent()" class="w-full py-2.5 bg-indigo-600 hover:bg-indigo-500 rounded-lg font-medium text-sm transition cursor-pointer">Scan for Fake / Clickbait</button>
                    <div id="result" class="text-xs bg-slate-950 p-3 rounded-lg border border-slate-800 hidden mt-2 text-slate-300"></div>
                `;
            }
        }

        function generateCreatorData() {
            const val = document.getElementById('userInput').value;
            const res = document.getElementById('result');
            if(!val) return alert("Kuch likhein pehle!");
            res.classList.remove('hidden');
            res.innerHTML = `🔥 **Hit Topic Idea:** Secret Mysteries of ${val}<br>✍️ **Title:** 99% log ye sach nahi jantay!<br>🎬 **Visual Guide:** Dark background with fast cinematic cuts.`;
        }

        function detectFakeContent() {
            const val = document.getElementById('userInput').value;
            const res = document.getElementById('result');
            if(!val) return alert("Kuch likhein pehle!");
            res.classList.remove('hidden');
            res.innerHTML = `⚠️ **Alert:** Yeh clickbait lag raha hai!<br>❌ **Verdict:** Time Waste / Fake Claims detected.<br>💡 **Advice:** Is video par apna time zaya mat karein.`;
        }

        function resetApp() {
            document.getElementById('workspace').classList.add('hidden');
            document.getElementById('gateway').classList.remove('hidden');
        }
    </script>
